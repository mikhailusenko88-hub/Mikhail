def forward(self,
    x: Tensor) -> Tensor:
  out = self.out
  x0 = torch.transpose((out).forward(x, ), 1, 2)
  mag, p, = torch.chunk(x0, 2, 1)
  mag0 = torch.exp(mag)
  mag1 = torch.clamp(mag0, None, 100.)
  x1 = torch.cos(p)
  y = torch.sin(p)
  S = torch.mul(mag1, torch.add(x1, torch.mul(y, 0.+1.j)))
  istft = self.istft
  return (istft).forward(S, )
